\section{}
\begin{enumerate}[(a)]
    \item Given $Z_1, \dots, Z_n$ are i.i.d. Bernoulli$(p)$ for $p \leq 1/2$
    and $t \in (0, 1 - p)$, we can compare the upper bound
    \[
        P\left[\frac{1}{n} \sum_{i=1}^{n} Z_i \geq t + p\right]
    \]
    between Hoeffding's inequality, Bernstein's inequality, and Bennett's inequality.
    First, define $X_i = Z_i - p$, consequently, $E[Z_i] = p, Var(Z_i) = p(1 - p)$,
    and thus $E[X_i] = 0, Var(X_i) = p(1 - p)$. By this same definition, we can derive
    that 
    \[
        P\left(\frac{1}{n} \sum_{i=1}^n Z_i \geq t + p \right) = P\left(\sum_{i=1}^n X_i \geq nt\right)
    \]
    Now for each inequality:
    \begin{itemize}
        \item \textbf{Hoeffding's inequality:} Hoeffding's
        inequality gives:
        \begin{gather*}
            P\left(\sum_{i=1}^{n} X_i \geq s) \leq \exp(-\frac{2s^2}{n})\right), \quad s = nt \therefore \\
            P\left(\frac{1}{n} \sum_{i=1}^{n} Z_i \geq t + p\right) \leq \exp(-2nt^2)
        \end{gather*}
        thus, $R_H = 2nt^2$.
        \item \textbf{Bernstein's inequality:} Using $X_i$, since $Z_i \in \{0, 1\}$,
        $X_i \in \{-p, 1 - p\}$. Because $p \leq 1/2$, $|X_i| \leq 1 - p$, and 
        that $\sum_{i=1}^n Var(X_i) = np(1 - p)$, Bernstein's inequality gives:
        \[
            P\left(\sum_{i=1}^{n} X_i \geq s\right) \leq \exp\left(-\frac{s^2}{2(v + bs/3)}\right)
        \]
        Again, let $s = nt$, with $v = np(1 - p)$ and $b = 1 - p$, we have:
        \begin{align*}
            P\left(\sum_{i=1}^{n} X_i \geq nt\right) &\leq \exp\left(-\frac{n^2 t^2}{2np(1 - p) + \frac{2}{3}(1 - p)nt}\right) \\
            &= \exp\left(-\frac{n^2t^2}{2n(1 - p)(p + t/3)}\right) \\
            &= \exp\left(-\frac{nt^2}{2(1 - p)(p + t/3)}\right)
        \end{align*}
        thus, $R_{\text{Bern}} = \frac{nt^2}{2(1 - p)(p + t/3)}$.
        \item \textbf{Bennett's inequality:} Using the same $X_i$ as above, Bennett's 
        inequality gives:
        \begin{align*}
            P\left(\sum_{i=1}^{n} X_i \geq s\right) &\leq \exp\left(-\frac{v}{b^2} h\left(\frac{bs}{v}\right)\right), \text{ using } s,v,b \therefore \\
            P\left(\sum_{i=1}^{n} X_i \geq nt\right) &\leq \exp\left(-\frac{np(1 - p)}{(1 - p)^2} h\left(\frac{(1 - p)nt}{np(1 - p)}\right)\right) \\
            &= \exp\left(-\frac{np}{1 - p} h\left(\frac{t}{p}\right)\right)
        \end{align*}
        thus, $R_{\text{Benn}} = \frac{np}{1 - p} h\left(\frac{t}{p}\right)$.
    \end{itemize}
    \item We can compute $R_{\text{Bern}}/R_H$ using $(a)$:
    \[
        \frac{R_{\text{Bern}}}{R_H} = \frac{\frac{nt^2}{2(1 - p)(p + t/3)}}{2nt^2} = \frac{1}{4(1 - p)(p + t/3)}
    \]
    We can see if Bernstein's bound improves over Hoeffding's more than a constant
    factor if $R_{\text{Bern}}/R_H \to \infty$ as $n \to \infty$. Since $p \leq \frac{1}{2}$,
    $\frac{1}{2} \leq 1 - p \leq 1$, so $1 - p$ is bounded away from 0. For the
    ratio to approach infinity, we need $p + t/3 \to 0$ as $n \to \infty$. This
    only occurs when $p \to 0$ and $t \to 0$ as $n \to \infty$. Thus, Bernstein's 
    bound improves over Hoeffding's more than a constant factor when $p \to 0$ and 
    $t \to 0$ as $n \to \infty$.
    \item $R_{\text{Benn}}/R_{\text{Bern}}$ depends on $(p, t)$ only through $x = t/p$:
    \begin{align*}
        \frac{R_{\text{Benn}}}{R_{\text{Bern}}} &= \frac{\frac{np}{1 - p} h\left(\frac{t}{p}\right)}{\frac{nt^2}{2(1 - p)(p + t/3)}} = \frac{2p(p + t/3)}{t^2} h\left(\frac{t}{p}\right) \\
        &= \frac{2p(p + px/3)}{p^2 x^2} h(x) = \frac{2(1 + x/3)}{x^2} h(x)
    \end{align*}
    Bennett's bound is never than Bernstein's since this ratio is always greater than 1:
    \begin{align*}
        \frac{2(1 + x/3)}{x^2} h(x) &\geq 1 \\
        h(x) &\geq \frac{x^2}{2(1 + x/3)} \\
        \text{Let } f(x) &= h(x) - \frac{x^2}{2(1 + x/3)} \\
        h(x) &= (1 + x)\log(1 + x) - x \therefore \\
        h'(x) &= \log(1 + x), h''(x) = \frac{1}{1 + x} \therefore \\
        f''(x) &= \frac{1}{1 + x} - \frac{1}{(1 + x/3)^3} \geq 0 
    \end{align*}
    For all $x \geq 0$, $(1 + x/3)^3 \geq 1 + x$, so $f''(x) \geq 0$, then $f'(x) \geq 0$ 
    also. With $f(0) = 0, f'(0) = 0$, we have $f(x) \geq 0$ for all $x \geq 0$. 
    Therefore, $h(x) \geq \frac{x^2}{2(1 + x/3)}$, and substituting this into the ratio gives:
    \[
        \frac{2(1 + x/3)}{x^2} h(x) \geq \frac{2(1 + x/3)}{x^2} \cdot \frac{x^2}{2(1 + x/3)} = 1
    \]
    thus, Bennett's bound is never worse than Bernstein's. \\
    As $x \to 0$, the taylor expansion of $h(x)$ gives $h(x) = \frac{x^2}{2} - \frac{x^3}{6} + O(x^4)$, so
    \[
        \frac{2(1 + x/3)}{x^2} h(x) = \frac{2(1 + x/3)}{x^2} \left(\frac{x^2}{2} - \frac{x^3}{6} + O(x^4)\right) = 1 + \frac{x^2}{18} + O(x)
    \]
    Thus, as $x \to 0$, the ratio approaches 1, and Bennett's bound is asymptotically 
    equivalent to Bernstein's. \\
    As $x \to \infty$, the taylor expansion of $h(x)$ gives $h(x) = x\log(x) - x + O(\log(x)) \sim x\log(x)$, so
    \[
        \frac{2(1 + x/3)}{x^2} h(x) = \frac{2(1 + x/3)}{x^2} (x\log(x)) \sim \frac{2}{3}\log(x) = \infty
    \]
    Thus, as $x \to \infty$, the ratio approaches infinity, and Bennett's bound is 
    asymptotically better than Bernstein's.
\end{enumerate}

\section{}
We are given the $\psi_\alpha$ Orlicz norm of a random variable $X$ defined as:
\[
    ||X||_{\psi_\alpha} = \inf\left\{K > 0 : E\left[\psi_\alpha \left(\frac{|X|}{K}\right)\right] \leq 1\right\}
\]
where $\alpha > 1$ and $\psi_\alpha(x) = e^{x^\alpha} - 1$. Using this, the
Orlicz norm can be rewritten as:
\[
    ||X||_{\psi_\alpha} = \inf\left\{K > 0 : E\left[e^{|X|^\alpha/K^\alpha}\right] \leq 2\right\}
\]
Suppose $||X||_{\psi_\alpha} < \infty$. By the definition of the infimum,
there exists a finite $K > 0$ where $E[e^{|X|^\alpha/K^\alpha}] \leq 2$. Then, 
for any $t > 0$, using Markov's inequality to the nonnegative random variable,
we get that $Y = e^{|X|^\alpha/K^\alpha}$. This gives:
\begin{align*}
    P(|X| \geq t) &= P(e^{|X|^\alpha/K^\alpha} \geq e^{t^\alpha/K^\alpha}) \\
    &\leq \frac{E[e^{|X|^\alpha/K^\alpha}]}{e^{t^\alpha/K^\alpha}} \\
    &\leq 2 e^{-t^\alpha/K^\alpha}
\end{align*}
Choosing $C = 2$ and $c = 1/K^\alpha$, we have that $P(|X| \geq t) \leq C e^{-ct^\alpha}$.
Thus, $||X||_{\psi_\alpha} < \infty$ implies that there exists a $C, c > 0$ such 
that $P(|X| \geq t) \leq C e^{-ct^\alpha}$. \\

\noindent
Now, we want to show that there exists a finite $K > 0$ such that $E[e^{|X|^\alpha/K^\alpha}] \leq 2$ 
if there exists a $C, c > 0$ such that $P(|X| \geq t) \leq C e^{-ct^\alpha}$. Let
$Y = |X|^\alpha \geq 0$, then for any $u \geq 0$:
\begin{align*}
    P(Y \geq u) &= P(|X|^\alpha \geq u) = P(|X| \geq u^{1/\alpha}) \\
    &\leq C e^{-c(u^{1/\alpha})} = C e^{-c u}
\end{align*}
For any $a > 0$, $e^{aY} = 1 + \int_0^Y a e^{au} du$. The tail integration identity
for nonnegative random variables gives us that $E[e^{aY}] = 1 + \int_0^\infty a e^{au} P(Y \geq u) du$. 
Using the tail bound on $P(Y \geq u)$, we have:
\begin{align*}
    E[e^{aY}] &\leq 1 + \int_0^\infty a e^{au} C e^{-cu} du \\
    &= 1 + aC \int_0^\infty e^{-(c - a)u} du \\
    &= 1 + \frac{aC}{c - a}
\end{align*}
Then if we choose $a = \frac{c}{C + 1}$, since $C, c > 0$, $0 < a < c$, and we have:
\begin{align*}
    E[e^{aY}] &\leq 1 + \frac{C(\frac{c}{C + 1})}{c - \frac{c}{C + 1}} \\
    &= 1 + \frac{\frac{Cc}{C + 1}}{\frac{cC}{C + 1}} \\
    &= 1 + 1 = 2
\end{align*}
Thus, $E[e^{aY}] \leq 2$. Recall that $Y = |X|^\alpha$, so we choose $K > 0$
such that $\frac{1}{K^\alpha} = a = \frac{c}{C + 1}$, and solving for $K$
gives us $K = \left(\frac{C + 1}{c}\right)^{1/\alpha}$. Then we have that
$E[e^{|X|^\alpha/K^\alpha}] \leq 2$, and then by the definition of the Orlicz norm,
there exists a finite $K > 0$. Because both implications hold, we have that:
\[
    ||X||_{\psi_\alpha} < \infty \iff \exists C, c > 0 : P(|X| \geq t) \leq C e^{-ct^\alpha} \quad \forall t \geq 0
\]

\section{}
From the definiton of the $\psi_\alpha$ Orlicz norm, we have that:
\[
    ||X||_{\psi_\alpha} = \inf\left\{K > 0 : E\left[e^{|X|^\alpha/K^\alpha}\right] \leq 2\right\}
\]
Define:
\[
    \beta = \frac{\alpha}{\alpha - 1} > 2 \therefore \frac{1}{\alpha} + \frac{1}{\beta} = 1, \beta(\alpha - 1) = \alpha
\]
By taking the Taylor expansion of the MGF $M(\lambda) = E[e^{\lambda Xi}]$, we have:
\begin{align*}
    e^{\lambda X_i} &\leq 1 + \lambda X_i + \frac{\lambda^2 |X_i|^2}{2} e^{|\lambda X_i|} \\
    E[e^{\lambda X_i}] &\leq 1 + \lambda E[X_i] + \frac{\lambda^2}{2} E[|X_i|^2 e^{|\lambda X_i|}] \\
    &\leq 1 + \frac{\lambda^2}{2} E[|X_i|^2 e^{|\lambda X_i|}], \lambda \in [0, 1] \therefore e^{\lambda X_i} \leq e^{|X_i|} \therefore \\
    E[e^{\lambda X_i}] &\leq 1 + \frac{\lambda^2}{2} E[X_i^2 e^{|X_i|}]
\end{align*}
We now need to bound $E[X_i^2 e^{|X_i|}]$. We want to find constant $C > 0$ such
that $X_i^2 e^{|X_i|} \leq C e^{|X_i|^\alpha/K^\alpha}$, rearrangeing gives:
\[
    X_i^2 e^{|X_i| - |X_i|^\alpha/K^\alpha} \leq C
\]
Because $\alpha > 1$, we have that $|X_i| - |X_i|^\alpha/K^\alpha \to -\infty$ as 
$|X_i| \to \infty$, then $X_i^2 e^{|X_i| - |X_i|^\alpha/K^\alpha} \to 0$ as 
$|X_i| \to \infty$. Then, there exists some constant $C > 0$ such that 
$X_i^2 e^{|X_i| - |X_i|^\alpha/K^\alpha} \leq C$ for all $|X_i| \geq 0$. Then, we 
have that:
\begin{align*}
    E[X_i^2 e^{|X_i|}] &\leq C E[e^{|X_i|^\alpha/K^\alpha}] \\
    &\leq 2C, \therefore \\
    E[e^{\lambda X_i}] &\leq 1 + \frac{\lambda^2}{2} 2C \leq e^{C\lambda^2}, \lambda \in [0, 1] \\
    \log E[e^{\lambda X_i}] &\leq C \lambda^2, \lambda \in [0, 1] \\
\end{align*}

\noindent
Now we bound the MGF for $\lambda > 1$. Using Young's inequality, we have:
\[
    ab \leq \frac{a^\alpha}{\alpha} + \frac{b^\beta}{\beta}
\]
Choosing $a = \frac{\alpha^{1/\alpha}|X_i|}{K}$ and $b = \frac{\lambda K}{\alpha^{1/\alpha}}$, 
we have:
\begin{align*}
    ab &= \lambda |X_i| \\
    \lambda |X_i| &\leq \frac{|X_i|^\alpha}{K^\alpha} + \frac{\lambda^\beta K^\beta}{\beta \alpha^{\beta/\alpha}} \\
\end{align*}
Let $C = \frac{K^\beta}{\beta \alpha^{\beta/\alpha}}$, then we have:
\begin{align*}
    \lambda |X_i| &\leq \frac{|X_i|^\alpha}{K^\alpha} + C \lambda^\beta \\
    e^{\lambda |X_i|} &\leq e^{|X_i|^\alpha/K^\alpha} e^{C \lambda^\beta} \\
    E[e^{\lambda |X_i|}] &\leq E[e^{|X_i|^\alpha/K^\alpha}] e^{C \lambda^\beta} \\
    &\leq 2 e^{C \lambda^\beta}, \text{ collapse constants}\therefore \\
    \log E[e^{\lambda |X_i|}] &\leq (\log 2 + C) \lambda^\beta, \lambda > 1, \beta = \frac{\alpha}{\alpha - 1}, \therefore \\
    \log E[e^{\lambda X_i}] &\leq C \lambda^{\alpha/(\alpha - 1)}, \lambda > 1 \\
\end{align*}

\noindent
Now, we can combine the two bounds on the MGF for $\lambda \in [0, 1]$ and $\lambda > 1$,
and establish some constant $C > 0$ that depends only on $K, \alpha$, such that:
\[
    \log E[e^{\lambda X_i}] \leq C(\lambda^2 + \lambda^\beta) \forall \lambda \geq 0
\]
Let $S_n = \sum_{i=1}^n X_i$, then by Markov's inequality, and $X_i$ independence, 
we have:
\begin{align*}
    P\left(\frac{S_n}{n} \geq t\right) &= P\left(e^{\lambda S_n} \geq e^{\lambda nt}\right) \\
    &\leq \frac{E[e^{\lambda S_n}]}{e^{\lambda nt}} \\
    \log E[e^{\lambda S_n}] &= \log E[e^{\lambda \sum_{i=1}^n X_i}] = \log \prod_{i=1}^n E[e^{\lambda X_i}] = \sum_{i=1}^n \log E[e^{\lambda X_i}] \\
    &\leq \sum_{i=1}^n C(\lambda^2 + \lambda^\beta) = nC(\lambda^2 + \lambda^\beta) \\
\end{align*}
Thus, we have:
\[
    P\left(\frac{S_n}{n} \geq t\right) \leq \exp(nC(\lambda^2 + \lambda^\beta) - \lambda nt)
\]
\begin{itemize}
    \item \textbf{$0 < t \leq 1$:}
    Choose some $\lambda = \delta t$, where $\delta > 0$ and is sufficiently small
    depending only on $C, \beta$. Then we have:
    \begin{align*}
        P\left(\frac{S_n}{n} \geq t\right) &\leq \exp(nC(\delta^2t^2 + \delta^\beta t^\beta) - \delta n t^2) \\
        &= \exp(n(C\delta^2 - \delta)t^2 + nC\delta^\beta t^\beta) \\
        &\beta > 2, 0 < t \leq 1, \therefore t^\beta \leq t^2, \\
        &\leq \exp(n(\delta^2 - C\delta^2 - C\delta^\beta)t^2), \text{ collapse constants} \\
        &\leq \exp(-c_1 nt^2), \text{ for some } c_1 > 0, \text{ sufficiently small } \delta
    \end{align*}
    \item \textbf{$t > 1$:}
    Choose some $\lambda = \delta t^{\alpha - 1}$, again with sufficiently small $\delta$:
    \begin{align*}
        P\left(\frac{S_n}{n} \geq t\right) &\leq \exp(nC(\delta^2 t^{2(\alpha - 1)} + \delta^\beta t^{\beta(\alpha - 1)}) - \delta n t^\alpha) \\
        &= \exp(n(C\delta^2 t^{2(\alpha - 1)} + C\delta^\beta t^\alpha) - \delta n t^\alpha) \\
        &= \exp(n(C\delta^2 t^{2(\alpha - 1)} + (C\delta^\beta - \delta) t^\alpha)) \\
        &t > 1, 2(\alpha - 1) < \alpha, \therefore t^{2(\alpha - 1)} < t^\alpha, \\
        &\leq \exp(n(C\delta^2 + C\delta^\beta - \delta)t^\alpha), \text{ collapse constants} \\
        &\leq \exp(-c_2 nt^\alpha), \text{ for some } c_2 > 0, \text{ sufficiently small } \delta
    \end{align*}
\end{itemize}

\noindent
Now combine both cases, and we have that for some constants $c_1, c_2 > 0$:
\begin{itemize}
    \item \textbf{For $0 < t \leq 1$:} $\min(t^2, t^\alpha) = t^2$
    \item \textbf{For $t > 1$:} $\min(t^2, t^\alpha) = t^\alpha$
\end{itemize}
\begin{align*}
    P\left(\frac{1}{n} \sum_{i=1}^n X_i \geq t\right) &\leq \begin{cases}
        e^{-cnt^2}, & 0 < t \leq 1 \\
        e^{-cnt^\alpha}, & t > 1
    \end{cases} \\
    P\left(\frac{1}{n} \sum_{i=1}^n X_i \geq t\right) &\leq C\exp\left(-cn\min(t^2, t^\alpha)\right), \quad \forall t > 0
\end{align*}

For any $G \sim \mathcal{N}(0, I_n)$ and for any fixed $v \in \mathbb{R}^n$,
$E_G[e^{v^\top G}] = e^{||v||_2^2/2}$. Conditioning on $X$, choose $v = \sqrt{2}\lambda B X$.
Since $BX \in \mathbb{R}^n$, the Gaussian MGF identity gives:
\begin{align*}
    E_G[e^{\sqrt{2}\lambda (BX)^\top G} | X] &= e^{||\sqrt{2}\lambda BX||_2^2/2} \\
    &= e^{\lambda^2 ||BX||_2^2} \\
    E_X[e^{\lambda^2 ||BX||_2^2}] &= E_X[E_G[e^{\sqrt{2}\lambda (BX)^\top G} | X]] \\
    &= E_G[E_X[e^{\sqrt{2}\lambda (BX)^\top G} | G]] \\
\end{align*}

\noindent
Since $X$ is $\sigma^2$-subgaussian, for any deterministic $u \in \mathbb{R}^d$,
$E[e^{u^\top X}] \leq e^{\sigma^2 ||u||_2^2/2}$. Conditioning on $G$, $B^\top G$
is fixed, so we can therefore choose $u = \sqrt{2}\lambda B^\top G$ and apply
the subgaussian property:
\begin{align*}
    E_X[e^{\sqrt{2}\lambda (BX)^\top G} | G] &= E_X[e^{\sqrt{2}\lambda X^\top B^\top G} | G] \\
    &\leq e^{\sigma^2 ||\sqrt{2}\lambda B^\top G||_2^2/2} \\
    &= e^{\sigma^2 \lambda^2 ||B^\top G||_2^2} \therefore \\
    E[e^{\lambda^2 ||BX||_2^2}] &\leq E_G[e^{\sigma^2 \lambda^2 ||B^\top G||_2^2}]
\end{align*}
which establishes the first inequality with $C = 1$. \\

Now we bound $E[e^{\lambda^2 \sigma^2 ||B^\top G||_2^2}]$. Let the SVD of $B$
be $B = U \Sigma V^\top$, where $U$ and $V$ are orthogonal matrices and $\Sigma$ 
is a diagonal matrix with singular values $s_1, \dots, s_r$ on the diagonal and 
$r = rank(B)$. Then we have:
\[
    B^\top G = V \Sigma^\top U^\top G, \text{ since } U^\top G \sim \mathcal{N}(0, I_n) \text{ and } V \text{ is orthogonal}
\]
Therefore,
\[
    ||B^\top G||_2^2 = ||\Sigma^\top U^\top G||_2^2 = \sum_{i=1}^r s_i^2 (U^\top G)_i^2 = \sum_{i=1}^r s_i^2 g_i^2
\]
Consequently by independence,
\[
    E[e^{\lambda^2 \sigma^2 ||B^\top G||_2^2}] = E[e^{\lambda^2 \sigma^2 \sum_{i=1}^r s_i^2 g_i^2}] = \prod_{i=1}^r E[e^{\lambda^2 \sigma^2 s_i^2 g_i^2}] \\
\]

\noindent
Now, for a standard Gaussian $g \sim \mathcal{N}(0, 1)$, we have that 
$E[e^{a g^2}] = (1 - 2a)^{-1/2}$ for $a < 1/2$. Let $a = \lambda^2 \sigma^2 s_i^2$, 
then we have:
\begin{align*}
    E[e^{\lambda^2 \sigma^2 s_i^2 g_i^2}] &= (1 - 2\lambda^2 \sigma^2 s_i^2)^{-1/2} \\
    E[e^{\lambda^2 \sigma^2 ||B^\top G||_2^2}] &= \prod_{i=1}^r (1 - 2\lambda^2 \sigma^2 s_i^2)^{-1/2} \\
    \log E[e^{\lambda^2 \sigma^2 ||B^\top G||_2^2}] &= -\frac{1}{2} \sum_{i=1}^r \log(1 - 2\lambda^2 \sigma^2 s_i^2)
\end{align*}
We can use the inequality $-\log(1 - u) \leq 2u$ for $u \in [0, 1/2]$, and choose
$\lambda$ such that $|\lambda| \leq \frac{1}{2\sigma ||B||_\text{op}}$, with
the fact that all $s_i \leq ||B||_\text{op}$, then 
$2\lambda^2 \sigma^2 s_i^2 \leq 1/2$ for all $i$. Then, we can collapse:
\begin{align*}
    \log E[e^{\lambda^2 \sigma^2 ||B^\top G||_2^2}] &\leq -\frac{1}{2} \sum_{i=1}^r 2(2\lambda^2 \sigma^2 s_i^2) \\
    &= \sum_{i=1}^r 2\lambda^2 \sigma^2 s_i^2 = 2\lambda^2 \sigma^2 ||B||_F^2 \\
    E[e^{\lambda^2 \sigma^2 ||B^\top G||_2^2}] &\leq e^{2\lambda^2 \sigma^2 ||B||_F^2}
\end{align*}
which establishes the second inequality with $C' = 2$. \\

\noindent
Thus, we can combine the inequalities to show:
\[
    E[e^{\lambda^2 ||BX||_2^2}] \leq E[e^{\lambda^2 \sigma^2 ||B^\top G||_2^2}] \leq e^{2\lambda^2 \sigma^2 ||B||_F^2}, \quad |\lambda| \leq \frac{1}{2\sigma ||B||_\text{op}}
\]
with constants $C = 1, C' = 2, c = \frac{1}{2}$. If $B = 0$ or $\sigma = 0$,
then the inequalities hold trivially (=1). \\
